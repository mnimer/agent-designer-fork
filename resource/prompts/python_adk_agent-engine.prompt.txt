2. **File Structure to Generate**:
   - `agent.py`: Reasoning Engine class defining the agent runtime:
     ```python
     import os
     from typing import Any, Dict, List, Optional
     from pydantic import BaseModel
     from pydantic_models import ResponseSchema

     class AgentReasoningEngine:
         """
         Vertex AI Agent Engine Reasoning Engine implementation.
         """
         def __init__(self, model_name: str = "gemini-2.0-flash", temperature: float = 0.7):
             self.model_name = model_name
             self.temperature = temperature
             self.system_prompt = "..."

         def set_up(self):
             """
             Initializes the client and resources when loaded inside Vertex AI Agent Engine.
             """
             from google import genai
             from google.genai import types
             # Uses Vertex AI with Application Default Credentials
             self.client = genai.Client(vertexai=True)

         def query(self, prompt: str, **kwargs) -> Dict[str, Any]:
             """
             Executes a single turn query against the agent.
             """
             from google.genai import types

             config = types.GenerateContentConfig(
                 system_instruction=self.system_prompt,
                 temperature=self.temperature,
                 response_mime_type="application/json",
                 response_schema=ResponseSchema,
             )

             response = self.client.models.generate_content(
                 model=self.model_name,
                 contents=prompt,
                 config=config,
             )

             if response.parsed and isinstance(response.parsed, ResponseSchema):
                 return response.parsed.model_dump()
             
             return {"raw_response": response.text}
     ```

    - `evaluate.py`: Vertex AI Agent Evaluation script utilizing the official `vertexai.evaluation` SDK and `EvalTask`:
      ```python
      import os
      import vertexai
      from vertexai.evaluation import EvalTask
      from agent import AgentReasoningEngine

      PROJECT_ID = os.getenv("GOOGLE_CLOUD_PROJECT", "YOUR_PROJECT_ID")
      LOCATION = os.getenv("GOOGLE_CLOUD_LOCATION", "us-central1")
      EXPERIMENT = os.getenv("VERTEX_EXPERIMENT", "agent-evaluation-experiment")

      vertexai.init(project=PROJECT_ID, location=LOCATION, experiment=EXPERIMENT)

      def evaluate_agent():
          """
          Runs comprehensive trajectory, groundedness, and instruction following evaluation.
          """
          print("Initializing Agent for Vertex AI Evaluation...")
          agent = AgentReasoningEngine()
          agent.set_up()

          dataset_path = os.path.join(os.path.dirname(__file__), "eval_dataset.jsonl")
          if not os.path.exists(dataset_path):
              print(f"Evaluation dataset not found at {dataset_path}")
              return

          eval_task = EvalTask(
              dataset=dataset_path,
              metrics=[
                  "trajectory_exact_match",
                  "trajectory_precision",
                  "response_follows_trajectory",
                  "groundedness",
                  "instruction_following",
                  "safety",
              ],
              experiment=EXPERIMENT,
          )

          print("Executing Vertex AI Agent Evaluation...")
          results = eval_task.evaluate(
              runnable=lambda query: agent.query(prompt=query) if isinstance(query, str) else agent.query(prompt=query.get("prompt", ""))
          )
          print("\n=== Vertex AI Evaluation Summary Metrics ===")
          print(results.summary_metrics)
          return results

      if __name__ == "__main__":
          evaluate_agent()
      ```
